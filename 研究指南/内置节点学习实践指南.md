   ```python
   def load_image(self, image, upload=None):
       # 1. 处理上传图像
       if upload is not None:
           return (upload,)
       
       # 2. 验证文件路径
       if not os.path.exists(image):
           raise FileNotFoundError(f"图像文件不存在: {image}")
       
       # 3. 加载图像
       img = Image.open(image)
       
       # 4. 转换为ComfyUI格式
       img_array = np.array(img)
       img_tensor = torch.from_numpy(img_array).float() / 255.0
       
       # 5. 添加批次维度
       if len(img_tensor.shape) == 3:
           img_tensor = img_tensor.unsqueeze(0)
       
       return (img_tensor,)
   ```

   ```python
   def encode(self, clip, text):
       # 1. 文本分词
       tokens = clip.tokenize(text)
       
       # 2. 编码文本
       cond, pooled = clip.encode_from_tokens(tokens, return_pooled=True)
       
       # 3. 创建条件
       conditioning = [[cond, {"pooled_output": pooled}]]
       
       return (conditioning,)
   ```

```python
# 创建文本分析节点
class TextAnalysisNode:
    @classmethod
    def INPUT_TYPES(cls):
        return {
            "required": {
                "text": ("STRING", {"multiline": True}),
                "clip": ("CLIP",)
            }
        }
    
    RETURN_TYPES = ("STRING", "INT")
    RETURN_NAMES = ("analysis", "token_count")
    FUNCTION = "analyze"
    CATEGORY = "Analysis"
    
    def analyze(self, text, clip):
        # 分词
        tokens = clip.tokenize(text)
        
        # 分析文本
        word_count = len(text.split())
        char_count = len(text)
        token_count = tokens.shape[1] if len(tokens.shape) > 1 else 0
        
        analysis = f"文本分析:\n"
        analysis += f"  字符数: {char_count}\n"
        analysis += f"  单词数: {word_count}\n"
        analysis += f"  Token数: {token_count}\n"
        analysis += f"  平均Token长度: {token_count/word_count:.2f}" if word_count > 0 else "  平均Token长度: 0"
        
        return (analysis, token_count)
```

```python
class KSampler:

                "steps": ("INT", {"default": 20, "min": 1, "max": 10000}),

                "sampler_name": (comfy.samplers.KSampler.SAMPLERS,),
                "scheduler": (comfy.samplers.KSampler.SCHEDULERS,),
                "positive": ("CONDITIONING",),
                "negative": ("CONDITIONING",),
                "model": ("MODEL",),
                "latent_image": ("LATENT",)
            }
        }
    
    RETURN_TYPES = ("LATENT",)
    RETURN_NAMES = ("LATENT",)
    FUNCTION = "sample"
    CATEGORY = "sampling"
```

#### 学习要点
1. **采样算法实现**
   ```python
   def sample(self, seed, steps, cfg, sampler_name, scheduler, positive, negative, model, latent_image):
       # 1. 设置随机种子
       torch.manual_seed(seed)
       
       # 2. 创建采样器
       sampler = comfy.samplers.KSampler(model, steps=steps, device=model.load_device)
       
       # 3. 设置调度器
       sampler.scheduler = comfy.samplers.KSampler.SCHEDULERS[scheduler]
       
       # 4. 执行采样
       samples = sampler.sample(
           steps=steps,
           cfg=cfg,
           sampler_name=sampler_name,
           positive=positive,
           negative=negative,
           latent_image=latent_image
       )
       
       return (samples,)
   ```

2. **性能优化技巧**
   - 合理设置步数
   - 优化CFG值
   - 使用合适的采样器

3. **内存管理**
   - 控制批次大小
   - 及时释放中间结果
   - 使用梯度检查点

```python
def process(self, input_data):
```
