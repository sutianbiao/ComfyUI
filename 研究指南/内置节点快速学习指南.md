## 🛠️ 实践练习

### 练习1：创建图像信息节点

```python
class ImageInfoNode:
    """显示图像信息"""
    @classmethod
    def INPUT_TYPES(cls):
        return {
            "required": {
                "image": ("IMAGE",)
            }
        }
    
    RETURN_TYPES = ("STRING",)
    RETURN_NAMES = ("info",)
    FUNCTION = "get_info"
    CATEGORY = "Analysis"
    
    def get_info(self, image):
        # 获取图像信息
        batch, height, width, channels = image.shape
        mean_brightness = torch.mean(image).item()
        
        info = f"图像信息:\n"
        info += f"  批次大小: {batch}\n"
        info += f"  尺寸: {width}x{height}\n"
        info += f"  通道数: {channels}\n"
        info += f"  平均亮度: {mean_brightness:.4f}\n"
        info += f"  数据类型: {image.dtype}\n"
        info += f"  设备: {image.device}"
        
        return (info,)
```

### 练习2：创建模型信息节点

```python
class ModelInfoNode:
    """显示模型信息"""
    @classmethod
    def INPUT_TYPES(cls):
        return {
            "required": {
                "model": ("MODEL",)
            }
        }
    
    RETURN_TYPES = ("STRING",)
    RETURN_NAMES = ("info",)
    FUNCTION = "get_info"
    CATEGORY = "Analysis"
    
    def get_info(self, model):
        # 获取模型信息
        total_params = sum(p.numel() for p in model.parameters())
        trainable_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
        
        info = f"模型信息:\n"
        info += f"  总参数数: {total_params:,}\n"
        info += f"  可训练参数: {trainable_params:,}\n"
        info += f"  设备: {next(model.parameters()).device}\n"
        info += f"  数据类型: {next(model.parameters()).dtype}"
        
        return (info,)
```

### 练习3：创建性能测试节点

```python
class PerformanceTestNode:
    """测试节点性能"""
    @classmethod
    def INPUT_TYPES(cls):
        return {
            "required": {
                "input_data": ("*",),
                "iterations": ("INT", {"default": 100, "min": 1, "max": 1000})
            }
        }
    
    RETURN_TYPES = ("STRING",)
    RETURN_NAMES = ("report",)
    FUNCTION = "test_performance"
    CATEGORY = "Testing"
    
    def test_performance(self, input_data, iterations):
        import time
        import psutil
        
        process = psutil.Process()
        start_memory = process.memory_info().rss / 1024 / 1024
        
        start_time = time.time()
        
        # 执行多次测试
        for i in range(iterations):
            # 模拟处理
            result = input_data * 2 if hasattr(input_data, '__mul__') else input_data
        
        end_time = time.time()
        end_memory = process.memory_info().rss / 1024 / 1024
        
        total_time = end_time - start_time
        avg_time = total_time / iterations
        memory_used = end_memory - start_memory
        
        report = f"性能测试报告:\n"
        report += f"  迭代次数: {iterations}\n"
        report += f"  总时间: {total_time:.4f}秒\n"
        report += f"  平均时间: {avg_time:.4f}秒\n"
        report += f"  内存使用: {memory_used:.2f}MB\n"
        report += f"  吞吐量: {1/avg_time:.2f}次/秒"
        
        return (report,)
```
