```python
class DebugNode(io.ComfyNode):
    @classmethod
    def define_schema(cls):
        return io.Schema(
            node_id="DebugNode",
            display_name="调试节点",
            inputs=[io.String.Input("message")],
            outputs=[io.String.Output()]
        )
    
    @classmethod
    def execute(cls, message):
        print(f"调试信息: {message}")
        return io.NodeOutput(f"处理完成: {message}")
```

### 3. 性能分析
```python
import cProfile
import pstats

# 分析函数性能
profiler = cProfile.Profile()
profiler.enable()
# 执行你的代码
profiler.disable()
stats = pstats.Stats(profiler)
stats.sort_stats('cumulative').print_stats(10)
```

### 4. 内存监控
```python
import psutil
import torch

def monitor_memory():
    process = psutil.Process()
    print(f"系统内存: {process.memory_info().rss / 1024 / 1024:.2f} MB")
    if torch.cuda.is_available():
        print(f"GPU内存: {torch.cuda.memory_allocated() / 1024 / 1024:.2f} MB")
```
