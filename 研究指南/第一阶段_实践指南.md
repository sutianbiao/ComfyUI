```python
# 创建 debug_utils.py
import logging
import time
import functools
import traceback
from pathlib import Path

class DebugLogger:
    def __init__(self, log_file="debug_logs/phase1_debug.log"):
        self.log_file = Path(log_file)
        self.log_file.parent.mkdir(exist_ok=True)
        
        # 设置日志格式
        logging.basicConfig(
            level=logging.DEBUG,
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
            handlers=[
                logging.FileHandler(self.log_file),
                logging.StreamHandler()
            ]
        )
        self.logger = logging.getLogger('ComfyUI_Debug')
    
    def log_function_call(self, func_name, args=None, kwargs=None):
        """记录函数调用"""
        self.logger.debug(f"调用函数: {func_name}")
        if args:
            self.logger.debug(f"  位置参数: {args}")
        if kwargs:
            self.logger.debug(f"  关键字参数: {kwargs}")
    
    def log_function_result(self, func_name, result):
        """记录函数结果"""
        self.logger.debug(f"函数 {func_name} 返回: {type(result)} - {str(result)[:100]}")
    
    def log_error(self, func_name, error):
        """记录错误"""
        self.logger.error(f"函数 {func_name} 发生错误: {error}")
        self.logger.error(f"错误堆栈: {traceback.format_exc()}")

# 全局调试器实例
debug_logger = DebugLogger()

def debug_function(func):
    """函数调试装饰器"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        debug_logger.log_function_call(func.__name__, args, kwargs)
        start_time = time.time()
        
        try:
            result = func(*args, **kwargs)
            debug_logger.log_function_result(func.__name__, result)
            return result
        except Exception as e:
            debug_logger.log_error(func.__name__, e)
            raise
        finally:
            end_time = time.time()
            debug_logger.logger.debug(f"函数 {func.__name__} 执行时间: {end_time - start_time:.4f}秒")
    
    return wrapper
```

```python
# 创建内存监控脚本
import psutil
import time
import threading

class MemoryMonitor:
    def __init__(self, interval=1.0):
        self.interval = interval
        self.monitoring = False
        self.data = []
    
    def start_monitoring(self):
        """开始监控"""
        self.monitoring = True
        self.monitor_thread = threading.Thread(target=self._monitor_loop)
        self.monitor_thread.daemon = True
        self.monitor_thread.start()
    
    def stop_monitoring(self):
        """停止监控"""
        self.monitoring = False
        if hasattr(self, 'monitor_thread'):
            self.monitor_thread.join()
    
    def _monitor_loop(self):
        """监控循环"""
        while self.monitoring:
            process = psutil.Process()
            memory_info = process.memory_info()
            
            self.data.append({
                'timestamp': time.time(),
                'rss': memory_info.rss / 1024 / 1024,  # MB
                'vms': memory_info.vms / 1024 / 1024,  # MB
                'cpu_percent': process.cpu_percent()
            })
            
            time.sleep(self.interval)
    
    def get_report(self):
        """获取监控报告"""
        if not self.data:
            return "没有监控数据"
        
        rss_values = [d['rss'] for d in self.data]
        vms_values = [d['vms'] for d in self.data]
        cpu_values = [d['cpu_percent'] for d in self.data]
        
        report = f"""
=== 内存监控报告 ===
监控时长: {len(self.data)} 秒
RSS内存: 最小={min(rss_values):.2f}MB, 最大={max(rss_values):.2f}MB, 平均={sum(rss_values)/len(rss_values):.2f}MB
VMS内存: 最小={min(vms_values):.2f}MB, 最大={max(vms_values):.2f}MB, 平均={sum(vms_values)/len(vms_values):.2f}MB
CPU使用: 最小={min(cpu_values):.2f}%, 最大={max(cpu_values):.2f}%, 平均={sum(cpu_values)/len(cpu_values):.2f}%
        """
        return report

# 使用示例
```

```python
import cProfile
import pstats
import io
from contextlib import contextmanager

@contextmanager
def profile_function(func_name):
    """函数性能分析上下文管理器"""
    profiler = cProfile.Profile()
    profiler.enable()
    
    try:
        yield profiler
    finally:
        profiler.disable()
        
        # 生成报告
        s = io.StringIO()
        ps = pstats.Stats(profiler, stream=s).sort_stats('cumulative')
        ps.print_stats(20)  # 显示前20个函数
        
        print(f"=== {func_name} 性能分析 ===")
        print(s.getvalue())
```
